CREATE DATABASE ECORECICLAJE
GO

USE ECORECICLAJE
GO


/* TABLA DE USUARIOS */

CREATE TABLE USUARIO
(
    ID_USUARIO INT IDENTITY(1,1) PRIMARY KEY,
    CEDULA VARCHAR(20) NOT NULL,
    NOMBRE VARCHAR(50) NOT NULL,
    CORREO VARCHAR(100) NOT NULL
)
GO


/* TABLA DE MATERIALES */

CREATE TABLE TIPO_MATERIAL
(
    ID_TIPO INT IDENTITY(1,1) PRIMARY KEY,
    NOMBRE VARCHAR(50) NOT NULL,
    DESCRIPCION VARCHAR(150)
)
GO


/* TABLA DE RECICLAJE */

CREATE TABLE RECICLAJE
(
    ID_RECICLAJE INT IDENTITY(1,1) PRIMARY KEY,
    ID_USUARIO INT NOT NULL,
    ID_TIPO INT NOT NULL,
    CANTIDAD DECIMAL(10,2) NOT NULL,
    FECHA DATE NOT NULL,

    FOREIGN KEY (ID_USUARIO)
        REFERENCES USUARIO(ID_USUARIO),

    FOREIGN KEY (ID_TIPO)
        REFERENCES TIPO_MATERIAL(ID_TIPO)
)
GO

/* PROCEDIMIENTOS DE USUARIOS */

CREATE PROCEDURE sp_listausuarios
AS
BEGIN
    SELECT
        ID_USUARIO,
        CEDULA,
        NOMBRE,
        CORREO
    FROM USUARIO
END
GO


CREATE PROCEDURE sp_agregarusuario
    @CEDULA VARCHAR(20),
    @NOMBRE VARCHAR(100),
    @CORREO VARCHAR(100)
AS
BEGIN
    INSERT INTO USUARIO
    (CEDULA, NOMBRE, CORREO)
    VALUES
    (@CEDULA, @NOMBRE, @CORREO)
END
GO


CREATE PROCEDURE sp_actualizarusuario
    @ID INT,
    @CEDULA VARCHAR(20),
    @NOMBRE VARCHAR(100),
    @CORREO VARCHAR(100)
AS
BEGIN
    UPDATE USUARIO
    SET
        CEDULA = @CEDULA,
        NOMBRE = @NOMBRE,
        CORREO = @CORREO
    WHERE ID_USUARIO = @ID
END
GO


CREATE PROCEDURE sp_borrarusuario
    @ID INT
AS
BEGIN
    DELETE FROM USUARIO
    WHERE ID_USUARIO = @ID
END
GO


/* PROCEDIMIENTO DE MATERIALES */

CREATE PROCEDURE sp_listatipos
AS
BEGIN
    SELECT
        ID_TIPO,
        NOMBRE,
        DESCRIPCION
    FROM TIPO_MATERIAL
END
GO


/* PROCEDIMIENTOS DE RECICLAJE */

CREATE PROCEDURE sp_agregarreciclaje
    @ID_USUARIO INT,
    @ID_TIPO INT,
    @CANTIDAD DECIMAL(10,2),
    @FECHA DATE
AS
BEGIN
    INSERT INTO RECICLAJE
    (ID_USUARIO, ID_TIPO, CANTIDAD, FECHA)
    VALUES
    (@ID_USUARIO, @ID_TIPO, @CANTIDAD, @FECHA)
END
GO


CREATE PROCEDURE sp_listareciclaje
AS
BEGIN
    SELECT
        R.ID_RECICLAJE,
        U.NOMBRE AS USUARIO,
        T.NOMBRE AS MATERIAL,
        R.CANTIDAD,
        R.FECHA
    FROM RECICLAJE R
    INNER JOIN USUARIO U
        ON R.ID_USUARIO = U.ID_USUARIO
    INNER JOIN TIPO_MATERIAL T
        ON R.ID_TIPO = T.ID_TIPO
END
GO


CREATE PROCEDURE sp_estadisticareciclaje
AS
BEGIN
    SELECT
        T.NOMBRE AS MATERIAL,
        SUM(R.CANTIDAD) AS TOTAL_KG
    FROM RECICLAJE R
    INNER JOIN TIPO_MATERIAL T
        ON R.ID_TIPO = T.ID_TIPO
    GROUP BY T.NOMBRE
    ORDER BY TOTAL_KG DESC
END
GO


/* ELIMINAR RECICLAJES DE UN USUARIO */

CREATE PROCEDURE sp_borrarreciclajesusuario
    @ID_USUARIO INT
AS
BEGIN
    DELETE FROM RECICLAJE
    WHERE ID_USUARIO = @ID_USUARIO
END
GO
